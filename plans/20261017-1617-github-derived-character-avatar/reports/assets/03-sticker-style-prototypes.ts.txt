// Prototype v3: "sticker" chibi style (user reference). 32x32 art, 3-tone muted ramps, auto outline,
// sticker ring added at render time on a padded canvas.
export type Grid = string[][];
export const N = 32;
export const blank = (): Grid => Array.from({ length: N }, () => Array(N).fill('.'));
const inb = (x: number, y: number) => x >= 0 && y >= 0 && x < N && y < N;
export function rect(g: Grid, x: number, y: number, w: number, h: number, c: string) {
  for (let j = y; j < y + h; j++) for (let i = x; i < x + w; i++) if (inb(i, j)) g[j][i] = c;
}
/** Rect with highlight on top/left edge and shade on right/bottom edge. */
export function shaded(g: Grid, x: number, y: number, w: number, h: number, [hi, mid, lo]: string) {
  rect(g, x, y, w, h, mid);
  rect(g, x, y, 1, h, hi); rect(g, x, y, w, 1, hi);
  rect(g, x + w - 1, y, 1, h, lo); rect(g, x, y + h - 1, w, 1, lo);
}
export function px(g: Grid, c: string, pts: [number, number][]) { for (const [x, y] of pts) if (inb(x, y)) g[y][x] = c; }
/** Rows of chars at (x, y); space = keep. */
export function rows(g: Grid, x: number, y: number, rs: string[]) {
  rs.forEach((r, j) => [...r].forEach((c, i) => { if (c !== ' ' && inb(x + i, y + j)) g[y + j][x + i] = c; }));
}
/** Exterior outline (8-neighbour = chunky sticker look). */
export function outline(g: Grid): Grid {
  const out = g.map((r) => [...r]);
  for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
    if (g[y][x] !== '.') continue;
    for (let dy = -1; dy <= 1; dy++) for (let dx = -1; dx <= 1; dx++) {
      const c = g[y + dy]?.[x + dx];
      if (c !== undefined && c !== '.' && c !== 'o') out[y][x] = 'o';
    }
  }
  return out;
}

// Turned toward the viewer's LEFT (user example): face/visor on the left, back of head on the right,
// plume/hat tip/hood point trail right, weapon upright in the near (left) hand, shield slot on the far (right) arm.
function face(g: Grid, eyeY: number) {
  px(g, 'o', [[11, eyeY], [11, eyeY + 1], [15, eyeY], [15, eyeY + 1]]); // far eye near the edge
  px(g, 'p', [[10, eyeY + 2], [16, eyeY + 2]]);
  px(g, 's', [[9, eyeY + 1]]); // nose bump
}
function legsAndBoots(g: Grid, leg: string) {
  rect(g, 11, 25, 3, 3, leg); rect(g, 17, 25, 3, 3, leg);
  shaded(g, 10, 28, 4, 3, 'llL'); shaded(g, 16, 28, 4, 3, 'llL'); // toes point left
}

export function warrior(): Grid {
  const g = blank();
  rows(g, 17, 0, ['  rr   ', ' rrrR  ', '  rrRR ', '    RRR', '     RR']); // plume trails back
  shaded(g, 10, 3, 12, 12, 'kiI'); px(g, '.', [[10, 3], [21, 3], [10, 14]]);
  rect(g, 10, 7, 8, 1, 'y'); rect(g, 10, 8, 7, 1, 'o'); rect(g, 10, 9, 8, 1, 'Y'); px(g, 'y', [[17, 8]]); // visor on the front
  rect(g, 19, 6, 1, 7, 'I'); // back plate seam
  shaded(g, 11, 15, 10, 8, 'kiI');
  rect(g, 11, 14, 8, 1, 'o'); // neck shadow separates helm from chest
  shaded(g, 18, 14, 4, 3, 'iiI'); // far pauldron
  rect(g, 11, 22, 10, 1, 'L'); px(g, 'y', [[13, 22], [14, 22]]);
  shaded(g, 12, 23, 3, 4, 'rrR'); // tabard flap at the front
  legsAndBoots(g, 'I');
  rect(g, 21, 17, 2, 4, 'I'); rect(g, 21, 21, 2, 2, 'I'); // far arm
  shaded(g, 9, 14, 5, 3, 'kiI'); rect(g, 9, 17, 2, 2, 'i'); // near pauldron + arm
  // sword upright in the near fist, held clear of the body
  rect(g, 4, 6, 1, 11, 'k'); rect(g, 5, 5, 1, 12, 'i'); rect(g, 6, 6, 1, 11, 'I');
  rect(g, 2, 17, 7, 1, 'y'); rect(g, 2, 18, 7, 1, 'Y');
  shaded(g, 4, 19, 6, 3, 'iiI'); px(g, 'y', [[5, 22]]);
  return outline(g);
}

/** Demo gear layer: tier-2 shield on the far arm, seen at an angle (testing slot). */
export function warriorWithShield(): Grid {
  const g = warrior().map((r) => r.map((c) => c));
  rows(g, 19, 14, [
    '  yyyy   ', ' yiiiiy  ', 'yiiiiiiY ', 'yiiiiiIY ', 'yiiyiiIY ', 'yiyyyiIY ', 'yiiyiiIY ',
    'yiiyiiIY ', 'yiiiiIIY ', ' yiiiIY  ', ' yYIIIY  ', '  YYYY   ',
  ]);
  return outline(g.map((r) => r.map((c) => (c === 'o' ? '.' : c))).map((r, y) => r.map((c, x) => (warrior()[y][x] === 'o' && c === '.' ? 'o' : c))));
}

  rect(g, 14, 3, 5, 1, 'u'); rect(g, 13, 4, 6, 2, 'u'); rect(g, 12, 6, 8, 2, 'u');
  px(g, 'c', [[14, 3], [13, 4], [13, 5], [12, 6], [12, 7]]); px(g, 'U', [[18, 3], [18, 4], [18, 5], [19, 6], [19, 7]]);
  rect(g, 11, 8, 10, 1, 'y');
  rect(g, 7, 9, 16, 1, 'U'); rect(g, 7, 9, 15, 1, 'u');
  rect(g, 10, 10, 12, 5, 's'); rect(g, 10, 10, 12, 1, 'S');
  rect(g, 18, 10, 4, 5, 'e'); rect(g, 21, 10, 1, 5, 'E'); rect(g, 18, 10, 1, 5, 'E'); // hair at the back

  return outline(g);
}

export function archer(): Grid {

  px(g, 'f', [[12, 2], [10, 3], [11, 3], [10, 4], [10, 5]]); rect(g, 21, 5, 1, 10, 'G');
  // face opening on the left, back of hood on the right with inner shadow
  rect(g, 10, 6, 8, 8, 's'); rect(g, 10, 6, 8, 2, 'h'); rect(g, 10, 6, 8, 1, 'H'); px(g, 'h', [[17, 8]]);
  rect(g, 18, 6, 1, 8, 'G'); rect(g, 10, 4, 9, 2, 'g'); rect(g, 10, 5, 8, 1, 'G');
  face(g, 9);
  shaded(g, 9, 14, 14, 4, 'fgG'); px(g, '.', [[9, 17], [22, 17]]);
  shaded(g, 11, 18, 10, 6, 'fgG');

  rect(g, 11, 23, 10, 1, 'L'); px(g, 'y', [[12, 23], [13, 23]]);

    '     w', '    wW', '   wW ', '  wW  ', ' wW   ', ' wW   ', 'wW    ', 'wW    ', 'wW    ', 'wW    ', 'wW    ', 'wW    ',
