// Phase 00 prototype: 3 tier-0 class looks on 32x32, drawn with primitives, auto 1px outline.
// Output: grids (for phase 04 port), HTML sheet, IoU table.
export type Grid = string[][];
export const N = 32;
export const blank = (): Grid => Array.from({ length: N }, () => Array(N).fill('.'));
export function rect(g: Grid, x: number, y: number, w: number, h: number, c: string) {
  for (let j = y; j < y + h; j++) for (let i = x; i < x + w; i++) if (i >= 0 && j >= 0 && i < N && j < N) g[j][i] = c;
}
export function px(g: Grid, c: string, pts: [number, number][]) { for (const [x, y] of pts) g[y][x] = c; }
export function rows(g: Grid, x: number, y: number, rs: string[]) {
  rs.forEach((r, j) => [...r].forEach((c, i) => { if (c !== ' ') g[y + j][x + i] = c; }));
}
/** Exterior outline: transparent cell 4-adjacent to any opaque cell becomes 'o'. */
export function outline(g: Grid): Grid {
  const out = g.map((r) => [...r]);
  for (let y = 0; y < N; y++) for (let x = 0; x < N; x++) {
    if (g[y][x] !== '.') continue;
    const n = [[1, 0], [-1, 0], [0, 1], [0, -1]].some(([dx, dy]) => {
      const c = g[y + dy]?.[x + dx];
      return c !== undefined && c !== '.' && c !== 'o';
    });
    if (n) out[y][x] = 'o';
  }
  return out;
}

/** Shared base body (fill only). Head x12-19 y7-13, torso x12-19 y14-22, legs y23-29, feet y30. */
export function baseBody(g: Grid, cloth: string, shade: string) {
  rect(g, 12, 6, 8, 8, 's'); // head
  rect(g, 12, 6, 8, 2, 'h'); // hair fringe
  px(g, 'h', [[12, 8], [19, 8]]);
  px(g, 'o', [[14, 10], [17, 10]]); // eyes
  px(g, 'S', [[12, 13], [19, 13], [13, 13], [18, 13]]); // jaw shade
  rect(g, 12, 14, 8, 9, cloth); // torso
  rect(g, 12, 21, 8, 1, shade); // belt line
  rect(g, 10, 15, 2, 5, cloth); rect(g, 20, 15, 2, 5, cloth); // arms
  rect(g, 10, 20, 2, 2, 's'); rect(g, 20, 20, 2, 2, 's'); // hands
  rect(g, 13, 23, 2, 6, shade); rect(g, 17, 23, 2, 6, shade); // legs
  rect(g, 12, 29, 3, 2, 'l'); rect(g, 17, 29, 3, 2, 'l'); // tier-0 boots (leather)
}

export function warrior(): Grid {
  const g = blank();
  baseBody(g, 'r', 'R');
  // horned helm: covers hair, iron, horns flare out x10/x21 up to y2
  rect(g, 12, 4, 8, 4, 'i'); rect(g, 13, 3, 6, 1, 'i'); rect(g, 12, 8, 8, 1, 'I');
  px(g, 'w', [[11, 5], [10, 4], [10, 3], [20, 5], [21, 4], [21, 3]]); // horns (bone ~ wood tone)
  px(g, 'I', [[15, 5], [16, 5], [15, 6], [16, 6], [15, 7], [16, 7]]); // nasal guard
  // pauldrons: wide square shoulders x10-21
  rect(g, 10, 14, 3, 3, 'i'); rect(g, 19, 14, 3, 3, 'i'); rect(g, 10, 16, 3, 1, 'I'); rect(g, 19, 16, 3, 1, 'I');
  // greatsword in main hand: 3px blade x5-7 y3-18, guard x3-9 y19, grip y20-22, pommel y23
  rect(g, 5, 4, 3, 15, 'i'); px(g, 'i', [[6, 3]]); rect(g, 6, 4, 1, 15, 'I');
  rect(g, 3, 19, 7, 1, 'b'); rect(g, 6, 20, 1, 3, 'w'); px(g, 'b', [[6, 23]]);
  rect(g, 8, 20, 2, 2, 's'); // fist on grip
  return outline(g);
}

export function mage(): Grid {
  const g = blank();
  baseBody(g, 'v', 'V');
  // pointed hat: brim x9-21 y7, cone up to tip y0 bent right
  rect(g, 10, 7, 12, 1, 'V'); rect(g, 12, 5, 8, 2, 'v'); rect(g, 13, 3, 6, 2, 'v');
  rect(g, 14, 2, 4, 1, 'v'); rect(g, 15, 1, 3, 1, 'v'); px(g, 'v', [[17, 0], [18, 0]]);
  rect(g, 12, 6, 8, 1, 'y'); // hat band
  px(g, 'a', [[15, 4]]); // star
  // flared robe: trapezoid to ankles, hides leg gap
  rect(g, 12, 22, 8, 2, 'v'); rect(g, 11, 24, 10, 3, 'v'); rect(g, 10, 27, 12, 2, 'V');
  rect(g, 15, 14, 2, 13, 'V'); // robe seam
  // staff: 1px wood x6 y6-30, orb above head line x4-8 y1-5
  rect(g, 6, 6, 1, 25, 'w');
  rows(g, 4, 1, [' aaa ', 'aaxaa', 'aaaaa', ' aaa ', '  w  ']);
  rect(g, 7, 19, 3, 2, 's'); // hand on staff
  return outline(g);
}

export function archer(): Grid {
  const g = blank();
  baseBody(g, 'g', 'G');
  // hood with back peak + feather
  rect(g, 11, 5, 10, 3, 'g'); rect(g, 12, 4, 7, 1, 'g'); rect(g, 11, 8, 1, 6, 'G'); rect(g, 20, 8, 1, 6, 'G');
  px(g, 'G', [[19, 3], [20, 4], [21, 5]]); // peak trailing right
  px(g, 'r', [[13, 3], [12, 2], [12, 1], [11, 0]]); // feather (red)
  // short cape/collar asymmetric, lean torso: belt + strap
  px(g, 'l', [[13, 15], [14, 16], [15, 17], [16, 18], [17, 19], [18, 20]]); // quiver strap
  // bow arc in main hand x2-7 y6-28, string x7
  rows(g, 2, 6, [
    '   ww', '  ww ', ' ww  ', ' w   ', 'ww   ', 'w    ', 'w    ', 'w    ', 'w    ', 'w    ', 'w    ',
    'w    ', 'w    ', 'w    ', 'w    ', 'w    ', 'w    ', 'ww   ', ' w   ', ' ww  ', '  ww ', '   ww',
  ]);
  for (let y = 7; y < 27; y++) g[y][7] = g[y][7] === '.' ? 'I' : g[y][7];
  rect(g, 8, 17, 2, 2, 's'); px(g, 'w', [[7, 17], [7, 18]]); // grip hand
  return outline(g);
}
