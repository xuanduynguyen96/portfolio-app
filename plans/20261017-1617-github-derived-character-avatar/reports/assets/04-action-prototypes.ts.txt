// Prototype v9: action poses, 2 frames per class, facing the viewer's left.
import { blank, outline, px, rect, rows, shaded, type Grid } from './sticker-prototypes.ts';

function line(g: Grid, x0: number, y0: number, x1: number, y1: number, c: string) {
  const dx = Math.abs(x1 - x0), dy = -Math.abs(y1 - y0), sx = x0 < x1 ? 1 : -1, sy = y0 < y1 ? 1 : -1;
  let err = dx + dy;
  for (;;) {
    if (g[y0]?.[x0] === '.') g[y0][x0] = c;
    if (x0 === x1 && y0 === y1) break;
    const e2 = 2 * err;
    if (e2 >= dy) { err += dy; x0 += sx; }
    if (e2 <= dx) { err += dx; y0 += sy; }
  }
}
function face(g: Grid, eyeY: number, dx = 0) {
  px(g, 'o', [[11 + dx, eyeY], [11 + dx, eyeY + 1], [15 + dx, eyeY], [15 + dx, eyeY + 1]]);
  px(g, 'p', [[10 + dx, eyeY + 2], [16 + dx, eyeY + 2]]);
  px(g, 's', [[9 + dx, eyeY + 1]]);
}

export function warrior(frame: 0 | 1): Grid {
  const g = blank();
  const b = frame; // frame 1 dips 1px deeper into the lunge
  rows(g, 16, 1 + b, [' rr   ', 'rrrR  ', ' rrRR ', '   RRR', '    RR']); // plume streams back
  shaded(g, 9, 4 + b, 12, 11, 'kiI'); px(g, '.', [[9, 4 + b], [20, 4 + b]]);
  rect(g, 9, 8 + b, 8, 1, 'y'); rect(g, 9, 9 + b, 7, 1, 'o'); rect(g, 9, 10 + b, 8, 1, 'Y');
  rect(g, 18, 7 + b, 1, 6, 'I');
  shaded(g, 10, 15 + b, 10, 8, 'kiI'); rect(g, 10, 14 + b, 7, 1, 'o');
  shaded(g, 17, 14 + b, 4, 3, 'iiI');
  rect(g, 10, 22 + b, 10, 1, 'L'); px(g, 'y', [[12, 22 + b], [13, 22 + b]]);
  shaded(g, 11, 23 + b, 3, 3 - b, 'rrR');
  // lunge: near leg bent forward, far leg stretched back
  rect(g, 9, 23 + b, 3, 2, 'I'); rect(g, 7, 25, 3, 3, 'I'); shaded(g, 4, 28, 5, 3, 'llL');
  rect(g, 16, 23 + b, 3, 3 - b, 'I'); rect(g, 18, 26, 3, 2, 'I'); shaded(g, 19, 28, 5, 3, 'llL');
  rect(g, 20, 17 + b, 2, 4, 'I'); rect(g, 20, 21 + b, 2, 2, 'I'); // far arm
  shaded(g, 8, 14 + b, 5, 3, 'kiI');
  if (frame === 0) {
    // en garde: blade angled up-forward
    for (let k = 0; k < 7; k++) { px(g, 'k', [[4 - k + 1, 14 - k]]); px(g, 'i', [[5 - k + 1, 14 - k]]); px(g, 'I', [[6 - k + 1, 14 - k]]); }
    px(g, 'k', [[0, 7]]);
    line(g, 4, 18, 9, 13, 'y'); px(g, 'y', [[4, 18], [5, 17], [6, 16], [7, 15], [8, 14], [9, 13]]);
    shaded(g, 6, 16, 4, 3, 'iiI'); px(g, 'y', [[10, 19]]);
  } else {
    // thrust: blade level, driven forward
    rect(g, 1, 16, 5, 1, 'k'); rect(g, 0, 17, 6, 1, 'i'); rect(g, 1, 18, 5, 1, 'I');
    rect(g, 6, 14, 1, 7, 'y'); rect(g, 6, 18, 1, 3, 'Y');
    shaded(g, 7, 16, 4, 3, 'iiI'); px(g, 'y', [[11, 17]]);
  }
  return outline(g);
}

export function mage(frame: 0 | 1): Grid {
  const g = blank();
  rows(g, 17, 0, ['    cu', '  cuuU', ' cuuU ']);
  rect(g, 14, 3, 5, 1, 'u'); rect(g, 13, 4, 6, 2, 'u'); rect(g, 12, 6, 8, 2, 'u');
  px(g, 'c', [[14, 3], [13, 4], [13, 5], [12, 6], [12, 7]]); px(g, 'U', [[18, 3], [18, 4], [18, 5], [19, 6], [19, 7]]);
  rect(g, 11, 8, 10, 1, 'y'); rect(g, 7, 9, 15, 1, 'u'); px(g, 'U', [[21, 9]]);
  rect(g, 10, 10, 12, 5, 's'); rect(g, 10, 10, 12, 1, 'S');
  rect(g, 18, 10, 4, 5, 'e'); rect(g, 18, 10, 1, 5, 'E'); rect(g, 21, 10, 1, 5, 'E');
  face(g, 11);
  // staff in the far hand, behind, crystal raised
  rect(g, 24, 6, 1, 23, 'w'); rect(g, 25, 6, 1, 23, 'W');
  rows(g, 22, 0, ['  aa  ', ' axaa ', ' aaaA ', '  aA  ', ' y  y ', '  yy  ']);
  if (frame === 1) px(g, 'x', [[22, 1], [27, 3]]);
  // robe swept back by the cast
  shaded(g, 11, 15, 10, 8, 'cuU');
  rect(g, 11, 22, 10, 1, 'L'); px(g, 'y', [[12, 22], [13, 22]]);
  shaded(g, 10, 23, 13, 4, 'cuU'); shaded(g, 10, 27, 15 + frame, 3, 'cuU'); rect(g, 10, 29, 15 + frame, 1, 'U');
  rect(g, 9, 30, 4, 1, 'L');
  rows(g, 8, 13, ['  eeeeeee ', 'eeeeeeeeeE', 'eeeeeeeeEE', ' eeeeeeeE ', '  eeeeeE  ', '  eeeE    ', '  eE      ']);
  px(g, 'E', [[12, 13], [13, 13]]);
  rect(g, 21, 15, 2, 3, 'U'); rect(g, 22, 17, 3, 2, 's'); // far arm on the staff
  // casting hand thrust forward + spell
  rect(g, 6, 15, 5, 2, 'u'); rect(g, 6, 15, 5, 1, 'c'); rect(g, 4, 15, 2, 2, 's');
  if (frame === 0) {
    rows(g, 0, 14, [' aa ', 'axaA', ' aA ']); px(g, 'x', [[2, 11], [0, 19]]);
  } else {
    rows(g, 0, 13, [' aa  ', 'aaxa ', 'axxaA', 'aaaA ', ' aA  ']); px(g, 'a', [[3, 10], [1, 20], [5, 12]]);
  }
  return outline(g);
}

export function archer(frame: 0 | 1): Grid {
  const g = blank();
  rows(g, 19, 0, ['  g', 'gGG']);
  rect(g, 12, 2, 8, 1, 'g'); rect(g, 10, 3, 11, 1, 'g'); rect(g, 10, 4, 12, 11, 'g');
  px(g, 'f', [[12, 2], [10, 3], [11, 3], [10, 4], [10, 5]]); rect(g, 21, 5, 1, 10, 'G');
  rect(g, 10, 6, 8, 8, 's'); rect(g, 10, 6, 8, 2, 'h'); rect(g, 10, 6, 8, 1, 'H'); px(g, 'h', [[17, 8]]);
  rect(g, 18, 6, 1, 8, 'G'); rect(g, 10, 4, 9, 2, 'g'); rect(g, 10, 5, 8, 1, 'G');
  face(g, 9);
  shaded(g, 9, 14, 14, 4, 'fgG'); px(g, '.', [[9, 17]]);
  rows(g, 21, 15, frame === 0 ? ['GG', 'G '] : ['GGG', 'GG ', 'G  ']); // cape flutters back
  shaded(g, 11, 18, 10, 6, 'fgG');
  px(g, 'L', [[19, 18], [18, 19], [17, 20], [16, 21], [15, 22], [14, 22]]);
  rect(g, 11, 23, 10, 1, 'L'); px(g, 'y', [[12, 23], [13, 23]]);
  // archer stance: near leg forward, far leg braced back
  rect(g, 10, 24, 3, 4, 'H'); shaded(g, 7, 28, 6, 3, 'llL');
  rect(g, 17, 24, 3, 3, 'H'); rect(g, 18, 27, 3, 1, 'H'); shaded(g, 18, 28, 5, 3, 'llL');
  // bow in the near hand, held out
  rows(g, 1, 3, [
    '     w', '    wW', '   wW ', '  wW  ', ' wW   ', ' wW   ', 'wW    ', 'wW    ', 'wW    ', 'wW    ', 'wW    ', 'wW    ',
    'wW    ', 'wW    ', 'wW    ', 'wW    ', 'wW    ', 'wW    ', 'wW    ', ' wW   ', ' wW   ', '  wW  ', '   wW ', '    wW',
    '     w',
  ]);
  rect(g, 7, 15, 3, 2, 'g'); rect(g, 7, 16, 3, 1, 'G'); rect(g, 5, 15, 2, 2, 'l'); rect(g, 3, 15, 2, 2, 's');
  // string drawn back to the anchor hand at the chest, arrow nocked (frame 1 = fuller draw)
  const hx = 11 + 2 * frame, ay = 14;
  line(g, 6, 4, hx, ay, 'e'); line(g, 6, 26, hx, ay, 'e');
  for (let x = 2; x < hx; x++) g[ay][x] = 'w';
  px(g, 'I', [[0, ay], [1, ay], [1, ay - 1], [1, ay + 1]]); px(g, 'e', [[hx - 1, ay - 1], [hx - 2, ay - 1], [hx - 1, ay + 1]]);
  rect(g, hx, ay, 2, 2, 's'); rect(g, hx + 2, ay, 3, 2, 'G'); // anchor hand + far forearm
  return outline(g);
}
